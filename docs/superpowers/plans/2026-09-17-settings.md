# Settings Page Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship the Settings page from `docs/superpowers/specs/2026-09-17-settings-design.md`: theme and reduce-motion preferences, one saved indexed folder, storage figures, and Clear index.

**Architecture:** Go owns `~/.spotfile/settings.json` through a small `engine/settings` package. The `App` exposes `GetSettings`, `SetAppearance` and `ClearIndex` and adds `folder` and `indexBytes` to `EngineStatus`. The frontend loads settings before mounting, derives the theme and motion attributes on `<html>` from a settings store, and renders a single-column Settings view built from small section and row components.

**Tech Stack:** Go 1.26, Wails v2.12, Svelte 3 + TypeScript, Vitest 0.34 (jsdom), svelte-spa-router.

**Branch:** `feat/settings-impl`, already created from `main`.

## Deviations from the spec (update the spec in Task 16)

1. **Package location:** the settings package lives at `engine/settings`, not `settings/`. CI only runs `go test ./engine/...`, and package `main` tests would need a built `frontend/dist`.
2. **Folder-reset logic:** it lives in `settings.WithFolder` (returns the new settings plus whether the folder changed). Clearing the index file lives in `vectorstore.Wipe` / `DiskSize`. That keeps the logic in CI-tested packages and leaves `App` thin.
3. **When a folder switch empties the index:** after the folder walk finds supported files. A folder with nothing to index therefore leaves the old index alone.
4. **Boot:** `main.ts` awaits `settings.load()` and then mounts, instead of calling a separate `applyAppearance`. The theme and motion stores stamp `<html>` during App init, before first paint.
5. **Wording:** the Library figures say "files", matching the existing status line, not "documents".

```go
// Package settings stores the user's preferences in a small JSON file.
package settings

import (
	"encoding/json"
	"errors"
	"fmt"
	"io/fs"
	"os"
	"path/filepath"
)

const (
	ThemeSystem   = "system"
	ThemeNocturne = "nocturne"
	ThemeDaylight = "daylight"

	MotionSystem = "system"
	MotionReduce = "reduce"
	MotionFull   = "full"
)

// Settings is everything Spotfile remembers between launches.
type Settings struct {
	Theme  string `json:"theme"`
	Motion string `json:"motion"`
	Folder string `json:"folder"` // the indexed folder; "" when none
}

// Defaults follow the OS for appearance and have no folder.
func Defaults() Settings {
	return Settings{Theme: ThemeSystem, Motion: MotionSystem}
}

// Validate reports an unknown theme or motion value.
func (s Settings) Validate() error {
	switch s.Theme {
	case ThemeSystem, ThemeNocturne, ThemeDaylight:
	default:
		return fmt.Errorf("unknown theme %q", s.Theme)
	}
	switch s.Motion {
	case MotionSystem, MotionReduce, MotionFull:
	default:
		return fmt.Errorf("unknown motion setting %q", s.Motion)
	}
	return nil
}

// WithFolder returns a copy that uses dir as the indexed folder, and whether
// dir is a different folder from the current one.
func (s Settings) WithFolder(dir string) (Settings, bool) {
	changed := s.Folder == "" || filepath.Clean(s.Folder) != filepath.Clean(dir)
	s.Folder = dir
	return s, changed
}

// Load reads settings from path. A missing file gives Defaults. A file that
// can't be used is renamed to path+".bad" and Defaults are returned with an
// error describing the problem, so a bad file never stops the app starting.
func Load(path string) (Settings, error) {
	data, err := os.ReadFile(path)
	if errors.Is(err, fs.ErrNotExist) {
		return Defaults(), nil
	}
	if err != nil {
		return Defaults(), setAside(path, err)
	}
	s := Defaults() // fields missing from the file keep their defaults
	if err := json.Unmarshal(data, &s); err != nil {
		return Defaults(), setAside(path, fmt.Errorf("decode settings: %w", err))
	}
	if err := s.Validate(); err != nil {
		return Defaults(), setAside(path, err)
	}
	return s, nil
}

func setAside(path string, cause error) error {
	if err := os.Rename(path, path+".bad"); err != nil {
		return fmt.Errorf("%w (moving the file aside also failed: %v)", cause, err)
	}
	return fmt.Errorf("%w (moved to %s.bad)", cause, filepath.Base(path))
}

// Save validates s and writes it to path via a temp file and rename, so a
// crash mid-write never leaves a truncated file.
func Save(path string, s Settings) error {
	if err := s.Validate(); err != nil {
		return err
	}
	data, err := json.MarshalIndent(s, "", "  ")
	if err != nil {
		return fmt.Errorf("encode settings: %w", err)
	}
	dir := filepath.Dir(path)
	if err := os.MkdirAll(dir, 0o755); err != nil {
		return fmt.Errorf("create settings dir: %w", err)
	}
	tmp, err := os.CreateTemp(dir, ".settings-*.tmp")
	if err != nil {
		return fmt.Errorf("create temp settings: %w", err)
	}
	tmpName := tmp.Name()
	defer os.Remove(tmpName) // no-op once the rename succeeds

	if _, err := tmp.Write(data); err != nil {
		tmp.Close()
		return fmt.Errorf("write settings: %w", err)
	}
	if err := tmp.Close(); err != nil {
		return fmt.Errorf("close temp settings: %w", err)
	}
	if err := os.Rename(tmpName, path); err != nil {
		return fmt.Errorf("replace settings: %w", err)
	}
	return nil
}
```

```go
// Wipe empties the store and deletes its persisted file. The store is emptied
// even when the file can't be removed.
func (vs *VectorStore) Wipe() error {
	vs.Clear()
	vs.mu.RLock()
	path := vs.persistPath
	vs.mu.RUnlock()
	if path == "" {
		return nil
	}
	if err := os.Remove(path); err != nil && !errors.Is(err, fs.ErrNotExist) {
		return fmt.Errorf("remove index file: %w", err)
	}
	return nil
}

// DiskSize is the size in bytes of the persisted file, or 0 when there is none.
func (vs *VectorStore) DiskSize() int64 {
	vs.mu.RLock()
	path := vs.persistPath
	vs.mu.RUnlock()
	if path == "" {
		return 0
	}
	info, err := os.Stat(path)
	if err != nil || info.IsDir() {
		return 0
	}
	return info.Size()
}
```

```go
package main

// Settings bindings: appearance preferences and the saved folder, persisted to
// ~/.spotfile/settings.json.

import (
	"fmt"
	"log"
	"os"
	"path/filepath"

	"spotfile/engine/settings"

	wailsruntime "github.com/wailsapp/wails/v2/pkg/runtime"
)

func settingsPath() string {
	return filepath.Join(spotfileDir(), "settings.json")
}

// loadSettings reads saved settings at startup. An unusable file is set aside
// and defaults are used, so settings never block launch.
func (a *App) loadSettings() {
	s, err := settings.Load(settingsPath())
	if err != nil {
		log.Printf("settings: %v; using defaults", err)
	}
	a.settingsMu.Lock()
	a.settings = s
	a.settingsMu.Unlock()
}

// GetSettings returns the current settings.
func (a *App) GetSettings() settings.Settings {
	a.settingsMu.Lock()
	defer a.settingsMu.Unlock()
	return a.settings
}

// SetAppearance saves the theme and reduce-motion preferences.
func (a *App) SetAppearance(theme, motion string) error {
	return a.updateSettings(func(s settings.Settings) settings.Settings {
		s.Theme = theme
		s.Motion = motion
		return s
	})
}

// updateSettings saves change(current) and keeps it only if the save succeeds.
func (a *App) updateSettings(change func(settings.Settings) settings.Settings) error {
	a.settingsMu.Lock()
	defer a.settingsMu.Unlock()
	next := change(a.settings)
	if err := settings.Save(settingsPath(), next); err != nil {
		return err
	}
	a.settings = next
	return nil
}

// useFolder records dir as the indexed folder. Switching to a different folder
// empties the index first, so the index only ever covers one folder.
func (a *App) useFolder(dir string) error {
	if _, changed := a.GetSettings().WithFolder(dir); !changed {
		return nil
	}
	a.store.Clear()
	if err := a.store.Persist(); err != nil {
		return fmt.Errorf("empty the index for the new folder: %w", err)
	}
	return a.updateSettings(func(s settings.Settings) settings.Settings {
		next, _ := s.WithFolder(dir)
		return next
	})
}

// resumeWatcher watches the saved folder again after a restart. A folder that
// no longer exists is forgotten.
func (a *App) resumeWatcher() {
	dir := a.GetSettings().Folder
	if dir == "" {
		return
	}
	if info, err := os.Stat(dir); err != nil || !info.IsDir() {
		log.Printf("settings: saved folder %q is gone; forgetting it", dir)
		forget := func(s settings.Settings) settings.Settings { s.Folder = ""; return s }
		if err := a.updateSettings(forget); err != nil {
			log.Printf("settings: %v", err)
		}
		return
	}
	if !a.indexMu.TryLock() {
		return // a folder index is starting and will start its own watcher
	}
	defer a.indexMu.Unlock()
	a.restartWatcher(dir)
}

// ClearIndex empties the search index and deletes its file. Models, the saved
// folder and the watcher are left alone.
// Emits: "index:cleared".
func (a *App) ClearIndex() error {
	if !a.indexMu.TryLock() {
		return fmt.Errorf("indexing is in progress")
	}
	defer a.indexMu.Unlock()
	err := a.store.Wipe()
	log.Printf("index: cleared by user")
	wailsruntime.EventsEmit(a.ctx, "index:cleared", nil)
	if err != nil {
		return fmt.Errorf("index cleared, but the file couldn't be removed: %w", err)
	}
	return nil
}
```

```ts
// Display formatting for figures and paths.

const STEP = 1024
const UNITS = ['KB', 'MB', 'GB']
const ELLIPSIS = '…'

// Bytes as a short size: "512 B", "2.1 MB". One decimal from KB up.
export function formatBytes(bytes: number): string {
  if (!Number.isFinite(bytes) || bytes < STEP) return `${Math.max(0, Math.round(bytes) || 0)} B`
  let value = bytes / STEP
  let unit = 0
  while (value >= STEP && unit < UNITS.length - 1) {
    value /= STEP
    unit++
  }
  return `${value.toFixed(1)} ${UNITS[unit]}`
}

// Shortens path to at most max characters by cutting from the middle, keeping
// the last segment (the folder's own name) whole when it fits.
export function truncateMiddle(path: string, max: number): string {
  if (path.length <= max) return path
  const cut = Math.max(path.lastIndexOf('/'), path.lastIndexOf('\\'))
  const tail = cut >= 0 ? path.slice(cut) : ''
  const room = max - tail.length - ELLIPSIS.length
  if (tail && room > 0) return path.slice(0, room) + ELLIPSIS + tail

  const keep = max - ELLIPSIS.length
  const head = Math.ceil(keep / 2)
  return path.slice(0, head) + ELLIPSIS + path.slice(path.length - (keep - head))
}
```

```ts
// Appearance preferences, saved by the Go backend in ~/.spotfile/settings.json.
// A change applies at once and rolls back if the save fails.

import { get, writable, type Readable } from 'svelte/store'

export type ThemePreference = 'system' | 'nocturne' | 'daylight'
export type MotionPreference = 'system' | 'reduce' | 'full'

export interface Appearance {
  theme: ThemePreference
  motion: MotionPreference
}

export interface SettingsState extends Appearance {
  error: string // last failed save, '' when none
}

export const defaultAppearance: Appearance = { theme: 'system', motion: 'system' }

// The backend surface the store needs; the app passes the Wails bindings.
export interface SettingsRuntime {
  get(): Promise<Appearance>
  setAppearance(theme: ThemePreference, motion: MotionPreference): Promise<void>
}

export interface SettingsStore extends Readable<SettingsState> {
  load(): Promise<void>
  setTheme(theme: ThemePreference): Promise<void>
  setMotion(motion: MotionPreference): Promise<void>
}

function messageOf(err: unknown): string {
  return err instanceof Error ? err.message : String(err)
}

export function createSettingsStore(runtime: SettingsRuntime): SettingsStore {
  const store = writable<SettingsState>({ ...defaultAppearance, error: '' })

  async function save<K extends keyof Appearance>(key: K, value: Appearance[K]): Promise<void> {
    const previous = get(store)[key]
    if (previous === value) return
    store.update((s) => ({ ...s, [key]: value }))
    const { theme, motion } = get(store)
    try {
      await runtime.setAppearance(theme, motion)
      store.update((s) => ({ ...s, error: '' }))
    } catch (err) {
      const error = `Couldn't save — ${messageOf(err)}`
      // Roll back only if no newer choice has replaced this value meanwhile.
      store.update((s) => (s[key] === value ? { ...s, [key]: previous, error } : { ...s, error }))
    }
  }

  return {
    subscribe: store.subscribe,
    async load() {
      try {
        const saved = await runtime.get()
        store.set({ theme: saved.theme, motion: saved.motion, error: '' })
      } catch (err) {
        console.error('settings: could not load, using defaults', err)
      }
    },
    setTheme: (theme) => save('theme', theme),
    setMotion: (motion) => save('motion', motion),
  }
}
```

```ts
// Resolves the theme preference against the OS light/dark appearance and
// stamps data-theme on the root element, which selects the Nocturne or
// Daylight tokens in tokens.css.

import { derived, readable, type Readable } from 'svelte/store'
import type { ThemePreference } from './settings'

export type ThemeName = 'nocturne' | 'daylight'

export interface ColorSchemeQuery {
  matches: boolean
  addEventListener(type: 'change', listener: (e: { matches: boolean }) => void): void
  removeEventListener(type: 'change', listener: (e: { matches: boolean }) => void): void
}

export function themeFor(preference: ThemePreference, prefersLight: boolean): ThemeName {
  if (preference !== 'system') return preference
  return prefersLight ? 'daylight' : 'nocturne'
}

// Whether the OS prefers light, updating live while subscribed.
function osPrefersLight(query: ColorSchemeQuery): Readable<boolean> {
  return readable(query.matches, (set) => {
    const onChange = (e: { matches: boolean }) => set(e.matches)
    set(query.matches)
    query.addEventListener('change', onChange)
    return () => query.removeEventListener('change', onChange)
  })
}

export function createThemeStore(
  query: ColorSchemeQuery,
  root: HTMLElement,
  preference: Readable<ThemePreference>,
): Readable<ThemeName> {
  return derived([preference, osPrefersLight(query)], ([pref, prefersLight]) => {
    const theme = themeFor(pref, prefersLight)
    root.dataset.theme = theme // applied here so every change reaches the DOM
    return theme
  })
}
```

```svelte
<script lang="ts">
  // One choice from a few options: a radiogroup where the arrow keys move the
  // selection, as native radio buttons do. Controlled: dispatches `change` and
  // waits for the parent to pass the new value back.
  import { createEventDispatcher } from 'svelte'

  export let label: string
  export let options: { value: string; label: string }[]
  export let value: string

  const dispatch = createEventDispatcher<{ change: string }>()
  const NEXT = new Set(['ArrowRight', 'ArrowDown'])
  const PREVIOUS = new Set(['ArrowLeft', 'ArrowUp'])
  let buttons: HTMLButtonElement[] = []

  function select(index: number) {
    buttons[index]?.focus()
    const chosen = options[index].value
    if (chosen !== value) dispatch('change', chosen)
  }

  function onKeydown(e: KeyboardEvent, index: number) {
    const step = NEXT.has(e.key) ? 1 : PREVIOUS.has(e.key) ? -1 : 0
    if (step === 0) return
    e.preventDefault()
    select((index + step + options.length) % options.length)
  }
</script>

      on:click={() => select(i)}
      on:keydown={(e) => onKeydown(e, i)}
```

```svelte
  // WebView. Controlled by `open`: Cancel and Esc dispatch `cancel`, and the
  // parent closes the dialog. Focus starts on Cancel so Enter never confirms.

  export let open = false

  export let confirmLabel: string
```
